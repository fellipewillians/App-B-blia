import React, { useState, useEffect, useRef } from 'react';
import { 
  Menu, X, Sun, Moon, Type, Plus, ChevronDown, 
  ChevronRight, Edit3, Bookmark, Check, Loader2,
  BookOpen, ChevronLeft, Trash2, ArrowRight, Sparkles
} from 'lucide-react';

const BIBLE_BOOKS = [
  { name: "Gênesis", id: "GEN", chapters: 50 }, { name: "Êxodo", id: "EXO", chapters: 40 },
  { name: "Levítico", id: "LEV", chapters: 27 }, { name: "Números", id: "NUM", chapters: 36 },
  { name: "Deuteronômio", id: "DEU", chapters: 34 }, { name: "Josué", id: "JOS", chapters: 24 },
  { name: "Juízes", id: "JDG", chapters: 21 }, { name: "Rute", id: "RUT", chapters: 4 },
  { name: "1 Samuel", id: "1SA", chapters: 31 }, { name: "2 Samuel", id: "2SA", chapters: 24 },
  { name: "1 Reis", id: "1KI", chapters: 22 }, { name: "2 Reis", id: "2KI", chapters: 25 },
  { name: "1 Crônicas", id: "1CH", chapters: 29 }, { name: "2 Crônicas", id: "2CH", chapters: 36 },
  { name: "Esdras", id: "EZR", chapters: 10 }, { name: "Neemias", id: "NEH", chapters: 13 },
  { name: "Ester", id: "EST", chapters: 10 }, { name: "Jó", id: "JOB", chapters: 42 },
  { name: "Salmos", id: "PSA", chapters: 150 }, { name: "Provérbios", id: "PRO", chapters: 31 },
  { name: "Eclesiastes", id: "ECC", chapters: 12 }, { name: "Cânticos", id: "SNG", chapters: 8 },
  { name: "Isaías", id: "ISA", chapters: 66 }, { name: "Jeremias", id: "JER", chapters: 52 },
  { name: "Lamentações", id: "LAM", chapters: 5 }, { name: "Ezequiel", id: "EZK", chapters: 48 },
  { name: "Daniel", id: "DAN", chapters: 12 }, { name: "Oseias", id: "HOS", chapters: 14 },
  { name: "Joel", id: "JOL", chapters: 3 }, { name: "Amós", id: "AMO", chapters: 9 },
  { name: "Obadias", id: "OBA", chapters: 1 }, { name: "Jonas", id: "JON", chapters: 4 },
  { name: "Miqueias", id: "MIC", chapters: 7 }, { name: "Naum", id: "NAM", chapters: 3 },
  { name: "Habacuque", id: "HAB", chapters: 3 }, { name: "Sofonias", id: "ZEP", chapters: 3 },
  { name: "Ageu", id: "HAG", chapters: 2 }, { name: "Zacarias", id: "ZEC", chapters: 14 },
  { name: "Malaquias", id: "MAL", chapters: 4 }, { name: "Mateus", id: "MAT", chapters: 28 },
  { name: "Marcos", id: "MRK", chapters: 16 }, { name: "Lucas", id: "LUK", chapters: 24 },

  { name: "Romanos", id: "ROM", chapters: 16 }, { name: "1 Coríntios", id: "1CO", chapters: 16 },

  { name: "Efésios", id: "EPH", chapters: 6 }, { name: "Filipenses", id: "PHP", chapters: 4 },
  { name: "Colossenses", id: "COL", chapters: 4 }, { name: "1 Tessalonicenses", id: "1TH", chapters: 5 },
  { name: "2 Tessalonicenses", id: "2TH", chapters: 3 }, { name: "1 Timóteo", id: "1TI", chapters: 6 },
  { name: "2 Timóteo", id: "2TI", chapters: 4 }, { name: "Tito", id: "TIT", chapters: 3 },
  { name: "Filemom", id: "PHM", chapters: 1 }, { name: "Hebreus", id: "HEB", chapters: 13 },
  { name: "Tiago", id: "JAS", chapters: 5 }, { name: "1 Pedro", id: "1PE", chapters: 5 },
  { name: "2 Pedro", id: "2PE", chapters: 3 }, { name: "1 João", id: "1JN", chapters: 5 },
  { name: "2 João", id: "2JN", chapters: 1 }, { name: "3 João", id: "3JN", chapters: 1 },
  { name: "Judas", id: "JUD", chapters: 1 }, { name: "Apocalipse", id: "REV", chapters: 22 }
];

export default function BibleApp() {
  // Navigation & Data State
  const [currentBook, setCurrentBook] = useState(BIBLE_BOOKS[0]);
  const [currentChapter, setCurrentChapter] = useState(1);
  const [chapterData, setChapterData] = useState([]);
  const [isLoadingChapter, setIsLoadingChapter] = useState(true);
  const [fetchError, setFetchError] = useState(false);
  const [view, setView] = useState('read'); // 'read' | 'notes'
  
  // Theme & Preferences
  const [darkMode, setDarkMode] = useState(() => {
    const saved = localStorage.getItem('bible-theme');
    return saved === 'dark' || (!saved && window.matchMedia('(prefers-color-scheme: dark)').matches);
  });
  const [fontSize, setFontSize] = useState(18);

  // UI State
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [expandedBookId, setExpandedBookId] = useState(BIBLE_BOOKS[0].id);
  const scrollRef = useRef(null);

  // 3D Intro Animation State
  const [showIntro, setShowIntro] = useState(true);
  const [introFinished, setIntroFinished] = useState(false);

  // Page Turn Animation State
  const [isAnimatingPage, setIsAnimatingPage] = useState(false);
  const [pageDirection, setPageDirection] = useState('forward');

  // User Data (Highlights & Notes)
  const [userData, setUserData] = useState(() => {
    const saved = localStorage.getItem('bible-user-data');
    return saved ? JSON.parse(saved) : {};
  });

  // Interaction State
  const [selectedVerse, setSelectedVerse] = useState(null);
  const [editingNoteFor, setEditingNoteFor] = useState(null);
  const [tempNote, setTempNote] = useState("");
  const popoverRef = useRef(null);

  // AI Panel State
  const [aiPanelOpen, setAiPanelOpen] = useState(false);
  const [aiLoading, setAiLoading] = useState(false);
  const [aiContent, setAiContent] = useState("");
  const [aiContext, setAiContext] = useState({ reference: "", text: "" });

  useEffect(() => {
    if (darkMode) {
      document.documentElement.classList.add('dark');
      localStorage.setItem('bible-theme', 'dark');
    } else {
      document.documentElement.classList.remove('dark');
      localStorage.setItem('bible-theme', 'light');
    }
  }, [darkMode]);

  useEffect(() => {
    localStorage.setItem('bible-user-data', JSON.stringify(userData));
  }, [userData]);

  useEffect(() => {
    // Fecha popover se clicar fora
    const handleClickOutside = (event) => {
      if (popoverRef.current && !popoverRef.current.contains(event.target)) {
        setSelectedVerse(null);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, []);

  // Animação de introdução
  useEffect(() => {
    const timer1 = setTimeout(() => setShowIntro(false), 800);
    const timer2 = setTimeout(() => setIntroFinished(true), 2000);
    return () => { clearTimeout(timer1); clearTimeout(timer2); };
  }, []);

  // Busca do capítulo atual sob demanda
  useEffect(() => {
    const fetchChapter = async () => {
      setIsLoadingChapter(true);
      setFetchError(false);
      try {
        const response = await fetch(`https://bible-api.com/${currentBook.id}+${currentChapter}?translation=almeida`);
        if (!response.ok) throw new Error("Erro na rede");
        const data = await response.json();
        setChapterData(data.verses || []);
      } catch (error) {
        console.error("Failed to fetch chapter:", error);
        setFetchError(true);
        setChapterData([]);
      } finally {
        setIsLoadingChapter(false);
      }
    };

    // Só busca depois da introdução ou se a intro já acabou
    if (introFinished) {
      fetchChapter();
    }
  }, [currentBook, currentChapter, introFinished]);

  const navigateToChapter = (book, chapterNum) => {
    if (currentBook.id === book.id && currentChapter === chapterNum) {
      setView('read');
      setSidebarOpen(false);
      return;
    }

    // Determina a direção para a animação
    const currentBookIndex = BIBLE_BOOKS.findIndex(b => b.id === currentBook.id);
    const targetBookIndex = BIBLE_BOOKS.findIndex(b => b.id === book.id);
    
    let direction = 'forward';
    if (targetBookIndex < currentBookIndex || (targetBookIndex === currentBookIndex && chapterNum < currentChapter)) {
      direction = 'backward';
    }

    // Inicia a animação de saída
    setPageDirection(direction);
    setIsAnimatingPage(true);
    
    setTimeout(() => {
      setCurrentBook(book);
      setCurrentChapter(chapterNum);
      setExpandedBookId(book.id);
      setView('read');
      setSidebarOpen(false);
      if (scrollRef.current) scrollRef.current.scrollTop = 0;
      
      // Remove a classe de animação após a mudança (permitindo que o novo conteúdo entre com fade/slide)
      setTimeout(() => {
        setIsAnimatingPage(false);
      }, 50);
    }, 300); // Duração da animação CSS de saída
  };

  const handlePrevChapter = () => {
    if (currentChapter > 1) {
      navigateToChapter(currentBook, currentChapter - 1);
    } else {
      const bookIndex = BIBLE_BOOKS.findIndex(b => b.id === currentBook.id);
      if (bookIndex > 0) {
        const prevBook = BIBLE_BOOKS[bookIndex - 1];
        navigateToChapter(prevBook, prevBook.chapters);
      }
    }
  };

  const handleNextChapter = () => {
    if (currentChapter < currentBook.chapters) {
      navigateToChapter(currentBook, currentChapter + 1);
    } else {
      const bookIndex = BIBLE_BOOKS.findIndex(b => b.id === currentBook.id);
      if (bookIndex < BIBLE_BOOKS.length - 1) {
        const nextBook = BIBLE_BOOKS[bookIndex + 1];
        navigateToChapter(nextBook, 1);
      }
    }
  };

  const highlightColors = [
    { id: 'yellow', class: 'bg-yellow-200 dark:bg-yellow-900/60' },
    { id: 'green', class: 'bg-green-200 dark:bg-green-900/60' },
    { id: 'pink', class: 'bg-pink-200 dark:bg-pink-900/60' },
  ];

  const handleHighlight = (color, verseObj) => {
    if (!selectedVerse) return;
    const key = `${currentBook.id}-${currentChapter}-${selectedVerse}`;
    setUserData(prev => {
      const newData = { ...prev };
      if (!newData[key]) {
        newData[key] = { 
          bookName: currentBook.name, bookId: currentBook.id, 
          chapter: currentChapter, verse: selectedVerse, text: verseObj.text 
        };
      }
      newData[key].highlight = color === newData[key].highlight ? null : color;
      
      if (!newData[key].highlight && !newData[key].note) delete newData[key];
      return newData;
    });
    setSelectedVerse(null);
  };

  const handleSaveNote = (verseKey, verseObj) => {
    setUserData(prev => {
      const newData = { ...prev };
      if (!newData[verseKey]) {
        newData[verseKey] = { 
          bookName: currentBook.name, bookId: currentBook.id, 
          chapter: currentChapter, verse: verseObj.verse, text: verseObj.text 
        };
      }
      newData[verseKey].note = tempNote;
      
      if (!newData[verseKey].highlight && !tempNote) delete newData[verseKey];
      return newData;
    });
    setEditingNoteFor(null);
    setTempNote("");
  };

  const handleDeleteItem = (key, type) => {
    setUserData(prev => {
      const newData = { ...prev };
      if (type === 'highlight') newData[key].highlight = null;
      if (type === 'note') newData[key].note = "";
      
      if (!newData[key].highlight && !newData[key].note) delete newData[key];
      return newData;
    });
  };

  const handleExplainWithAI = async (verseText, verseNum) => {
    setSelectedVerse(null);
    const reference = `${currentBook.name} ${currentChapter}:${verseNum}`;
    setAiPanelOpen(true);
    setAiLoading(true);
    setAiContent("");
    setAiContext({ reference, text: verseText });
    
    try {
      const prompt = `Atue como um teólogo e estudioso bíblico erudito. Explique o contexto histórico e o significado teológico do seguinte versículo da Bíblia de forma clara, moderna e acessível, em português do Brasil: [${reference}] "${verseText}". Formate a resposta em parágrafos curtos e diretos, sem marcações markdown excessivas.`;
      const payload = {
          contents: [{ parts: [{ text: prompt }] }]
      };
      
      const apiKey = ""; // Automático no ambiente Canvas
      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
      
      const response = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
      });
      const result = await response.json();
      const text = result?.candidates?.[0]?.content?.parts?.[0]?.text;
      
      if (text) {
         setAiContent(text);
      } else {
         setAiContent("Desculpe, a IA não conseguiu processar uma explicação neste momento. Tente novamente mais tarde.");
      }
    } catch (error) {
      console.error("Erro na API da IA:", error);
      setAiContent("Ocorreu um erro de conexão ao tentar buscar a explicação com a inteligência artificial. Verifique sua rede.");
    } finally {
      setAiLoading(false);
    }
  };

  if (!introFinished) {
    return (
      <div className="fixed inset-0 z-50 flex items-center justify-center bg-gray-950 perspective-container">
        <div className="absolute w-12 h-64 bg-black rounded-l-lg shadow-2xl z-0 transform -translate-x-1/2"></div>
        <div className={`
          relative w-48 sm:w-64 h-72 sm:h-96 bg-stone-800 rounded-r-2xl shadow-2xl flex flex-col items-center justify-center transform-gpu origin-left transition-all duration-[1200ms] ease-in-out z-10 border-l-4 border-stone-950
          ${showIntro ? 'rotate-y-0' : 'rotate-y-[-160deg] opacity-0'}
        `}>
          <div className="absolute inset-0 border-2 border-stone-700 m-2 rounded-r-xl opacity-50"></div>
          <BookOpen className="w-16 h-16 text-stone-300 mb-6 drop-shadow-lg" />
          <h1 className="text-2xl font-serif font-bold text-stone-200 text-center uppercase tracking-widest">Bíblia<br/>Sagrada</h1>
        </div>
        <div className="absolute w-48 sm:w-64 h-[17.5rem] sm:h-[23.5rem] bg-[#f8f5ee] rounded-r-xl shadow-inner ml-1 flex items-center justify-center overflow-hidden">
           <div className="text-stone-400 font-serif text-[10px] opacity-40 select-none px-4 text-justify line-clamp-[16]">
             No princípio, criou Deus os céus e a terra. A terra, porém, estava sem forma e vazia; havia trevas sobre a face do abismo, e o Espírito de Deus pairava por sobre as águas. Disse Deus: Haja luz; e houve luz. E viu Deus que a luz era boa; e fez separação entre a luz e as trevas.
           </div>
        </div>
        <style>{`
          .perspective-container { perspective: 1800px; }
          .rotate-y-0 { transform: rotateY(0deg); }
          .rotate-y-\\[-160deg\\] { transform: rotateY(-160deg); }
        `}</style>
      </div>
    );
  }

  return (
    <div className={`min-h-screen flex transition-colors duration-300 ${darkMode ? 'dark bg-[#111111] text-gray-200' : 'bg-[#ffffff] text-gray-800'}`}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600&family=Merriweather:ital,wght@0,300;0,400;0,700;1,300&display=swap');
        .font-sans { font-family: 'Inter', sans-serif; }
        .font-serif { font-family: 'Merriweather', serif; }
        
        .custom-scrollbar::-webkit-scrollbar { width: 4px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background: #cbd5e1; border-radius: 4px; }
        .dark .custom-scrollbar::-webkit-scrollbar-thumb { background: #333333; }

        /* Page Turn Animation Classes */
        .page-reading-area {
          transition: transform 0.3s cubic-bezier(0.4, 0, 0.2, 1), opacity 0.3s ease;
          transform-origin: center center;
        }
        .page-turning-forward { transform: translateX(-30px) rotateY(-15deg); opacity: 0; }
        .page-turning-backward { transform: translateX(30px) rotateY(15deg); opacity: 0; }
      `}</style>

      {/* Overlay Mobile Sidebar */}
      {sidebarOpen && (
        <div className="fixed inset-0 bg-black/40 z-40 lg:hidden backdrop-blur-sm transition-opacity" onClick={() => setSidebarOpen(false)} />
      )}

      {/* Sidebar de Navegação Completa (66 Livros) */}
      <aside className={`
        fixed lg:static inset-y-0 left-0 z-50 w-72 transform transition-transform duration-300 ease-in-out flex flex-col shadow-2xl lg:shadow-none
        ${sidebarOpen ? 'translate-x-0' : '-translate-x-full lg:translate-x-0'}
        ${darkMode ? 'bg-[#1a1a1a] border-gray-800' : 'bg-[#f7f7f5] border-gray-200'} border-r
      `}>
        <div className="p-6 pb-4 flex-shrink-0">
          <div className="flex items-center justify-between mb-6">
            <h1 className="text-xl font-bold tracking-tight font-sans flex items-center gap-2 text-stone-700 dark:text-stone-300">
              <BookOpen className="w-5 h-5 text-indigo-500" /> Estudo Bíblico
            </h1>
            <button className="lg:hidden p-2 rounded-md hover:bg-black/5 dark:hover:bg-white/5" onClick={() => setSidebarOpen(false)}>
              <X className="w-5 h-5" />
            </button>
          </div>
          
          <button 
            onClick={() => { setView('notes'); setSidebarOpen(false); }}
            className={`w-full flex items-center gap-2 py-2.5 px-4 mb-2 rounded-xl font-medium transition-all shadow-sm
              ${view === 'notes' ? 'bg-indigo-600 text-white' : darkMode ? 'bg-[#2a2a2a] hover:bg-[#333] text-gray-300' : 'bg-white hover:bg-gray-50 border border-gray-200 text-gray-700'}`}
          >
            <Bookmark className="w-4 h-4" /> Minhas Anotações
          </button>
        </div>
          
        <div className="flex-1 overflow-y-auto px-4 pb-6 custom-scrollbar font-sans text-sm">
          {BIBLE_BOOKS.map((book) => (
            <div key={book.id} className="mb-1">
              <button 
                onClick={() => setExpandedBookId(expandedBookId === book.id ? null : book.id)}
                className={`flex w-full items-center justify-between py-2.5 px-3 rounded-lg font-medium transition-colors
                  ${expandedBookId === book.id ? (darkMode ? 'bg-[#222]' : 'bg-gray-100/80') : 'hover:bg-black/5 dark:hover:bg-white/5'}
                `}
              >
                {book.name}
                {expandedBookId === book.id ? <ChevronDown className="w-4 h-4 text-gray-400" /> : <ChevronRight className="w-4 h-4 text-gray-400" />}
              </button>
              
              <div className={`
                grid grid-cols-5 gap-1 overflow-hidden transition-all duration-300 ease-in-out pl-1 pr-1
                ${expandedBookId === book.id ? 'max-h-[500px] opacity-100 py-2' : 'max-h-0 opacity-0'}
              `}>
                {Array.from({ length: book.chapters }, (_, i) => i + 1).map(chapterNum => {
                  const isCurrent = currentBook.id === book.id && currentChapter === chapterNum && view === 'read';
                  return (
                    <button
                      key={`${book.id}-${chapterNum}`}
                      onClick={() => navigateToChapter(book, chapterNum)}
                      className={`
                        py-1.5 rounded-md text-center transition-colors text-xs font-semibold
                        ${isCurrent ? 'bg-indigo-600 text-white shadow-sm' : darkMode ? 'hover:bg-[#333] text-gray-400' : 'hover:bg-gray-200 text-gray-600'}
                      `}
                    >
                      {chapterNum}
                    </button>
                  );
                })}
              </div>
            </div>
          ))}
        </div>
      </aside>

      {/* Main Content Area */}
      <main className="flex-1 flex flex-col h-screen overflow-hidden relative perspective-container">
        
        {/* Header Superior */}
        <header className={`
          flex items-center justify-between px-4 sm:px-8 py-4 z-30 relative flex-shrink-0 transition-colors
          ${darkMode ? 'bg-[#111111]/80 border-gray-800 backdrop-blur-md' : 'bg-white/80 border-gray-100 backdrop-blur-md border-b'}
        `}>
          <div className="flex items-center gap-3">
            <button className="lg:hidden p-2 -ml-2 rounded-md hover:bg-black/5 dark:hover:bg-white/5 transition-colors" onClick={() => setSidebarOpen(true)}>
              <Menu className="w-5 h-5" />
            </button>
            <div className="text-sm font-sans font-medium text-gray-500 dark:text-gray-400 uppercase tracking-wider select-none">
              {view === 'read' ? `${currentBook.name} ${currentChapter}` : 'Dashboard'}
            </div>
          </div>

          <div className="flex items-center gap-2 sm:gap-4">
            {view === 'read' && (
              <div className={`hidden sm:flex items-center rounded-full p-1 ${darkMode ? 'bg-[#222]' : 'bg-gray-100'}`}>
                <button onClick={() => setFontSize(Math.max(14, fontSize - 2))} className="p-1.5 rounded-full hover:bg-black/10 dark:hover:bg-white/10 transition-colors"><Type className="w-4 h-4" /></button>
                <span className="w-px h-4 bg-gray-300 dark:bg-gray-700 mx-1"></span>
                <button onClick={() => setFontSize(Math.min(32, fontSize + 2))} className="p-1.5 rounded-full hover:bg-black/10 dark:hover:bg-white/10 transition-colors"><Plus className="w-4 h-4" /></button>
              </div>
            )}

            <button onClick={() => setDarkMode(!darkMode)} className="p-2 rounded-full hover:bg-black/5 dark:hover:bg-white/5 transition-colors">
              {darkMode ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5 text-gray-600" />}
            </button>
          </div>
        </header>

        {/* Área Rolável de Leitura ou Dashboard */}
        <div ref={scrollRef} className="flex-1 overflow-y-auto px-4 md:px-12 py-10 scroll-smooth custom-scrollbar relative">
          
          {view === 'notes' ? (
            <div className="max-w-4xl mx-auto pb-20 animate-in fade-in duration-500">
              <div className="flex items-center gap-3 mb-10">
                <Bookmark className="w-8 h-8 text-indigo-500" />
                <h2 className="text-3xl font-serif font-bold text-gray-900 dark:text-white">Minhas Anotações</h2>
              </div>
              
              {Object.keys(userData).length === 0 ? (
                <div className="text-center py-24 text-gray-400 dark:text-gray-600 flex flex-col items-center">
                  <Bookmark className="w-16 h-16 mb-4 opacity-20" />
                  <p className="font-sans text-lg">Nenhum destaque ou nota encontrada.</p>
                  <p className="text-sm mt-2 font-serif">Selecione versículos durante a leitura para salvá-los aqui.</p>
                </div>
              ) : (
                <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                  {Object.entries(userData).map(([key, item]) => {
                    const bookData = BIBLE_BOOKS.find(b => b.id === item.bookId) || BIBLE_BOOKS[0];
                    return (
                      <div key={key} className={`p-6 rounded-2xl border transition-all hover:shadow-lg ${darkMode ? 'bg-[#1a1a1a] border-gray-800' : 'bg-white border-gray-100 shadow-sm'}`}>
                        <div className="flex justify-between items-start mb-4">
                          <button 
                            onClick={() => navigateToChapter(bookData, item.chapter)}
                            className="text-lg font-bold text-indigo-600 dark:text-indigo-400 hover:text-indigo-500 font-sans flex items-center gap-1 group"
                          >
                            {item.bookName} {item.chapter}:{item.verse}
                            <ArrowRight className="w-4 h-4 opacity-0 group-hover:opacity-100 transition-opacity transform group-hover:translate-x-1" />
                          </button>
                        </div>

                        {item.highlight && (
                          <div className="mb-4">
                            <div className="flex items-center gap-2 mb-2">
                              <span className={`w-3 h-3 rounded-full ${highlightColors.find(c=>c.id===item.highlight)?.class || 'bg-gray-200'}`}></span>
                              <span className="text-xs text-gray-400 font-sans uppercase tracking-wider font-semibold">Destaque</span>
                              <button onClick={() => handleDeleteItem(key, 'highlight')} className="text-gray-400 hover:text-red-500 transition-colors ml-auto"><Trash2 className="w-4 h-4" /></button>
                            </div>
                            <p className={`text-base font-serif italic p-3 rounded-lg ${highlightColors.find(c=>c.id===item.highlight)?.class || ''}`}>
                              "{item.text}"
                            </p>
                          </div>
                        )}

                        {item.note && (
                          <div>
                            <div className="flex items-center gap-2 mb-2 mt-4 pt-4 border-t border-gray-100 dark:border-gray-800">
                              <Edit3 className="w-4 h-4 text-gray-400" />
                              <span className="text-xs text-gray-400 font-sans uppercase tracking-wider font-semibold">Sua Nota</span>
                              <button onClick={() => handleDeleteItem(key, 'note')} className="text-gray-400 hover:text-red-500 transition-colors ml-auto"><Trash2 className="w-4 h-4" /></button>
                            </div>
                            <p className={`text-sm p-4 rounded-xl font-sans text-gray-700 dark:text-gray-300 ${darkMode ? 'bg-[#222]' : 'bg-gray-50'}`}>
                              {item.note}
                            </p>
                          </div>
                        )}
                      </div>
                    );
                  })}
                </div>
              )}
            </div>
          ) : (
            <div className={`
              max-w-3xl mx-auto pb-32 page-reading-area
              ${isAnimatingPage ? (pageDirection === 'forward' ? 'page-turning-forward' : 'page-turning-backward') : ''}
            `}>
              <h2 className="text-4xl md:text-5xl font-serif font-bold mb-14 text-center text-gray-900 dark:text-gray-100">
                {currentBook.name} {currentChapter}
              </h2>
              
              {isLoadingChapter ? (
                <div className="space-y-6 opacity-50 animate-pulse">
                  {[1, 2, 3, 4, 5].map(i => (
                    <div key={i} className="flex gap-4">
                      <div className={`h-4 w-4 rounded mt-2 ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                      <div className="flex-1 space-y-3">
                        <div className={`h-4 w-full rounded ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                        <div className={`h-4 w-5/6 rounded ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                      </div>
                    </div>
                  ))}
                </div>
              ) : fetchError ? (
                <div className="text-center py-10 text-red-500 font-sans bg-red-50 dark:bg-red-900/10 rounded-xl">
                  Falha ao carregar o capítulo. Verifique sua conexão.
                </div>
              ) : (
                <div className="space-y-7 font-serif leading-loose transition-all duration-300 text-gray-800 dark:text-gray-300" style={{ fontSize: `${fontSize}px` }}>
                  {chapterData.map((verseObj) => {
                    const verseNum = verseObj.verse;
                    const verseKey = `${currentBook.id}-${currentChapter}-${verseNum}`;
                    const uData = userData[verseKey] || {};
                    const isSelected = selectedVerse === verseNum;
                    
                    return (
                      <div key={verseNum} id={`verse-${verseNum}`} className="relative group">
                        <p 
                          className={`cursor-pointer rounded-xl p-3 -mx-3 transition-all duration-200 
                            ${isSelected ? (darkMode ? 'bg-[#222] shadow-sm' : 'bg-white shadow-lg border border-gray-100') : 'hover:bg-black/5 dark:hover:bg-white/5'}
                          `}
                          onClick={() => setSelectedVerse(isSelected ? null : verseNum)}
                        >
                          <sup className="text-[0.6em] font-sans text-indigo-400 dark:text-indigo-500 font-bold mr-3 top-[-0.5em] relative">{verseNum}</sup>
                          <span className={`${uData.highlight ? highlightColors.find(c => c.id === uData.highlight)?.class + ' rounded-md px-1 py-0.5' : ''}`}>
                            {verseObj.text}
                          </span>
                        </p>

                        {/* Popover de Ações do Versículo */}
                        {isSelected && (
                          <div ref={popoverRef} className={`absolute left-0 md:left-1/2 md:-translate-x-1/2 mt-3 z-20 flex items-center gap-1.5 p-2 rounded-2xl shadow-2xl border transition-all animate-in fade-in zoom-in duration-200 ${darkMode ? 'bg-[#2a2a2a] border-gray-700' : 'bg-white border-gray-100'}`}>
                            <div className="flex gap-1.5 border-r border-gray-200 dark:border-gray-700 pr-2">
                              {highlightColors.map(color => (
                                <button key={color.id} onClick={() => handleHighlight(color.id, verseObj)} className={`w-8 h-8 rounded-full ${color.class} border-2 hover:scale-110 transition-transform ${uData.highlight === color.id ? 'border-gray-800 dark:border-gray-200 scale-110' : 'border-transparent'}`} title="Destacar" />
                              ))}
                            </div>
                            <button onClick={() => { setEditingNoteFor(verseKey); setTempNote(uData.note || ""); setSelectedVerse(null); }} className={`flex items-center gap-2 px-4 py-2 text-sm font-sans font-semibold rounded-xl transition-colors ${darkMode ? 'hover:bg-[#333] text-gray-200' : 'hover:bg-gray-100 text-gray-700'}`}>
                              <Edit3 className="w-4 h-4" /> Anotar
                            </button>
                            <div className="w-px h-6 bg-gray-200 dark:bg-gray-700 mx-1"></div>
                            <button onClick={() => handleExplainWithAI(verseObj.text, verseNum)} className={`flex items-center gap-2 px-4 py-2 text-sm font-sans font-semibold rounded-xl transition-colors text-indigo-600 dark:text-indigo-400 hover:bg-indigo-50 dark:hover:bg-indigo-900/30`}>
                              <Sparkles className="w-4 h-4" /> Explicar com IA
                            </button>
                          </div>
                        )}

                        {/* Exibição / Edição da Nota Abaixo do Versículo */}
                        {(uData.note || editingNoteFor === verseKey) && (
                          <div className={`mt-4 ml-8 p-5 rounded-2xl font-sans text-base shadow-sm border ${darkMode ? 'bg-[#1a1a1a] border-gray-800' : 'bg-[#fafafa] border-gray-200'}`}>
                            {editingNoteFor === verseKey ? (
                              <div className="flex flex-col gap-4 animate-in fade-in">
                                <textarea value={tempNote} onChange={(e) => setTempNote(e.target.value)} placeholder="Escreva suas reflexões aqui..." autoFocus
                                  className={`w-full p-4 rounded-xl outline-none resize-y min-h-[120px] border transition-colors ${darkMode ? 'bg-[#111] border-gray-700 text-white focus:border-indigo-500' : 'bg-white border-gray-200 focus:border-indigo-400'}`} />
                                <div className="flex justify-end gap-3">
                                  <button onClick={() => setEditingNoteFor(null)} className={`px-5 py-2.5 text-sm rounded-xl transition-colors font-semibold ${darkMode ? 'hover:bg-[#333]' : 'hover:bg-gray-200'}`}>Cancelar</button>
                                  <button onClick={() => handleSaveNote(verseKey, verseObj)} className="px-5 py-2.5 text-sm bg-indigo-600 hover:bg-indigo-700 text-white font-semibold rounded-xl transition-colors flex items-center gap-2"><Check className="w-4 h-4" /> Salvar Nota</button>
                                </div>
                              </div>
                            ) : (
                              <div className="group relative cursor-pointer" onClick={() => { setEditingNoteFor(verseKey); setTempNote(uData.note); }}>
                                <p className="leading-relaxed whitespace-pre-wrap text-gray-600 dark:text-gray-400 pr-6">{uData.note}</p>
                                <div className="absolute top-0 right-0 p-1 opacity-0 group-hover:opacity-100 transition-opacity bg-gray-100 dark:bg-[#333] rounded-md">
                                  <Edit3 className="w-4 h-4 text-indigo-500" />
                                </div>
                              </div>
                            )}
                          </div>
                        )}
                      </div>
                    );
                  })}
                </div>
              )}

              {/* Botões de Navegação Inferior */}
              {!isLoadingChapter && !fetchError && (
                <div className="mt-20 pt-10 border-t border-gray-200 dark:border-gray-800 flex items-center justify-between font-sans">
                  <button onClick={handlePrevChapter} className={`flex items-center gap-2 px-5 py-3 rounded-xl transition-colors font-semibold ${darkMode ? 'bg-[#1a1a1a] hover:bg-[#222]' : 'bg-gray-50 hover:bg-gray-100 border border-gray-200'}`}>
                    <ChevronLeft className="w-5 h-5" /> Anterior
                  </button>
                  <div className="text-sm text-gray-400 hidden sm:block uppercase tracking-widest font-semibold">Fim de {currentBook.name}</div>
                  <button onClick={handleNextChapter} className={`flex items-center gap-2 px-5 py-3 rounded-xl transition-colors font-semibold ${darkMode ? 'bg-[#1a1a1a] hover:bg-[#222]' : 'bg-gray-50 hover:bg-gray-100 border border-gray-200'}`}>
                     Próximo <ChevronRight className="w-5 h-5" />
                  </button>
                </div>
              )}
            </div>
          )}
        </div>

        {aiPanelOpen && (
          <div className="fixed inset-0 bg-black/40 z-50 transition-opacity backdrop-blur-sm lg:hidden" onClick={() => setAiPanelOpen(false)}></div>
        )}

        <div className={`
          fixed top-0 right-0 h-full w-full sm:w-[28rem] md:w-[32rem] z-50 transform transition-transform duration-500 ease-out shadow-2xl flex flex-col
          ${aiPanelOpen ? 'translate-x-0' : 'translate-x-full'}
          ${darkMode ? 'bg-[#111111] border-l border-gray-800' : 'bg-white border-l border-gray-200'}
        `}>
          <div className={`p-6 border-b flex items-center justify-between flex-shrink-0 ${darkMode ? 'border-gray-800' : 'border-gray-100'}`}>
            <h3 className="font-bold font-sans flex items-center gap-2 text-indigo-600 dark:text-indigo-400 text-xl">
              <Sparkles className="w-6 h-6" /> Explicador IA
            </h3>
            <button onClick={() => setAiPanelOpen(false)} className="p-2.5 rounded-full hover:bg-black/5 dark:hover:bg-white/5 transition-colors">
              <X className="w-5 h-5" />
            </button>
          </div>
          
          <div className="p-8 overflow-y-auto custom-scrollbar flex-1">
            <div className={`p-5 rounded-2xl mb-8 font-serif italic text-base border-l-4 border-indigo-500 shadow-sm ${darkMode ? 'bg-[#1a1a1a] text-gray-300' : 'bg-gray-50 text-gray-700'}`}>
              <div className="font-bold font-sans not-italic text-xs text-indigo-500 mb-3 uppercase tracking-widest">
                {aiContext.reference}
              </div>
              "{aiContext.text}"
            </div>

            {aiLoading ? (
              <div className="space-y-5 animate-pulse mt-10">
                <div className={`h-4 rounded-md w-3/4 ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                <div className={`h-4 rounded-md w-full ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                <div className={`h-4 rounded-md w-5/6 ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                <div className={`h-4 rounded-md w-full ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                <div className={`h-4 rounded-md w-4/6 ${darkMode ? 'bg-gray-800' : 'bg-gray-200'}`}></div>
                
                <div className="flex items-center justify-center gap-2 text-indigo-500 mt-12 text-sm font-semibold animate-bounce">
                  <Sparkles className="w-4 h-4" /> Analisando teologia e contexto histórico...
                </div>
              </div>
            ) : (
              <div className="font-sans text-base leading-loose whitespace-pre-wrap text-gray-700 dark:text-gray-300">
                {aiContent}
              </div>
            )}
          </div>
        </div>

      </main>
    </div>
  );
}
